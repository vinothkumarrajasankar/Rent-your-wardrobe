Supabase Database Schema & Setup Guide

Run the following SQL script directly in your Supabase SQL Editor. It creates the profiles, wardrobe_items, and rentals tables along with production Row-Level-Security (RLS) policies and automatic profile creation triggers: 

-- 1. Create Profiles Table (extends Supabase auth.users)
create table if not exists public.profiles (
  id uuid references auth.users on delete cascade primary key,
  full_name text,
  phone text,
  address text,
  is_admin boolean default false,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Create Wardrobe Items Table
create table if not exists public.wardrobe_items (
  id uuid default gen_random_uuid() primary key,
  title text not null,
  brand text not null,
  size text not null,
  category text not null,
  rental_price_4d numeric not null check (rental_price_4d >= 0),
  rental_price_8d numeric not null check (rental_price_8d >= 0),
  deposit_amount numeric not null check (deposit_amount >= 0),
  image_url text not null,
  is_active boolean default true,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 3. Create Rentals Table
create table if not exists public.rentals (
  id uuid default gen_random_uuid() primary key,
  item_id uuid references public.wardrobe_items on delete cascade not null,
  renter_id uuid references auth.users on delete set null not null,
  start_date date not null,
  end_date date not null,
  buffer_end_date date not null,
  total_amount numeric not null check (total_amount >= 0),
  deposit_amount numeric not null check (deposit_amount >= 0),
  status text not null default 'confirmed' check (status in ('pending', 'confirmed', 'dispatched', 'returned', 'completed', 'cancelled')),
  deposit_status text not null default 'held' check (deposit_status in ('held', 'refunded', 'forfeited')),
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 4. Enable Row Level Security (RLS)
alter table public.profiles enable row level security;
alter table public.wardrobe_items enable row level security;
alter table public.rentals enable row level security;

-- 5. RLS Policies
-- Profiles
create policy "Users can view own profile" on public.profiles for select using (auth.uid() = id);
create policy "Admins can view all profiles" on public.profiles for select using (
  exists (select 1 from public.profiles where id = auth.uid() and is_admin = true)
);
create policy "Users can update own profile" on public.profiles for update using (auth.uid() = id);
create policy "Users can insert own profile" on public.profiles for insert with check (auth.uid() = id);

-- Wardrobe Items
create policy "Public can view active items" on public.wardrobe_items for select using (
  is_active = true or exists (select 1 from public.profiles where id = auth.uid() and is_admin = true)
);
create policy "Admins can manage wardrobe items" on public.wardrobe_items for all using (
  exists (select 1 from public.profiles where id = auth.uid() and is_admin = true)
);

-- Rentals
create policy "Renters can view own rentals" on public.rentals for select using (
  auth.uid() = renter_id or exists (select 1 from public.profiles where id = auth.uid() and is_admin = true)
);
create policy "Renters can create rentals" on public.rentals for insert with check (auth.uid() = renter_id);
create policy "Admins can update rentals" on public.rentals for update using (
  exists (select 1 from public.profiles where id = auth.uid() and is_admin = true)
);

-- 6. Trigger to automatically generate a profile on Supabase auth user signup
create or replace function public.handle_new_user()
returns trigger as $$
begin
  insert into public.profiles (id, full_name, phone, address, is_admin)
  values (new.id, coalesce(new.raw_user_meta_data->>'full_name', 'Fashion Enthusiast'), '', '', false);
  return new;
end;
$$ language plpgsql security definer;

create or replace trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();
